L = N - len(h) + 1

H = FFT(h)
for (t = 0; t < len(x) - N; t += L) {
        
    # len(xt) = N
    xt = x[t..t + N]
    
    # len(yt) = N
    # inconsistency here: either h or H needs to be zero-padded to length N
    yt = IFFT(FFT(xt) * H)

    # len(yt) = L
    yt = yt[len(h)..N]
    
    y[t..t + L] = yt
}
